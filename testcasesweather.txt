import React from "react";
import { render, screen, fireEvent } from "@testing-library/react";
import App from "./App";

describe("App Component", () => {
  test("renders Search component", () => {
    render(<App />);
    const searchElement = screen.getByPlaceholderText(/Search for city/i);
    expect(searchElement).toBeInTheDocument();
  });

  test("renders toggle button", () => {
    render(<App />);
    const toggleButton = screen.getByText(/Toggle to °F/i);
    expect(toggleButton).toBeInTheDocument();
  });

  test("toggles temperature unit", () => {
    render(<App />);
    const toggleButton = screen.getByText(/Toggle to °F/i);
    fireEvent.click(toggleButton);
    expect(toggleButton).toHaveTextContent(/Toggle to °C/i);
  });
});


import React from "react";
import { render, screen, fireEvent } from "@testing-library/react";
import Search from "./components/search/search";

describe("Search Component", () => {
  test("renders search input", () => {
    render(<Search onSearchChange={jest.fn()} />);
    const searchInput = screen.getByPlaceholderText(/Search for city/i);
    expect(searchInput).toBeInTheDocument();
  });

  test("handles search input change", () => {
    const handleOnSearchChange = jest.fn();
    render(<Search onSearchChange={handleOnSearchChange} />);
    const searchInput = screen.getByPlaceholderText(/Search for city/i);

    fireEvent.change(searchInput, { target: { value: "New York" } });
    expect(searchInput.value).toBe("New York");
  });
});


import React from "react";

import React from "react";
import { render, screen } from "@testing-library/react";
import HourlyWeather from "./components/hourly-weather/hourly-weather";

const mockData = {
  list: [
    {
      dt: new Date().getTime() / 1000, // current date time in seconds
      main: { temp: 15 },
      weather: [{ description: "cloudy", icon: "03d" }],
    },
  ],
};

describe("HourlyWeather Component", () => {
  test("renders hourly forecast", () => {
    render(<HourlyWeather data={mockData} unit="C" convertTemperature={(temp) => temp} />);
    
    const currentHour = new Date().getHours();
    expect(screen.getByText(`${currentHour}:00`)).toBeInTheDocument();
    expect(screen.getByText(/cloudy/i)).toBeInTheDocument();
    expect(screen.getByText(/15°C/i)).toBeInTheDocument();
  });
});
