import React from "react";
import "./hourly-weather.css";

const HourlyWeather = ({ data, unit, convertTemperature }) => {
  const getHourlyData = (hour) => {
    // Filter and format hourly data for the current day
    const hourlyList = data.list.filter((item) => new Date(item.dt * 1000).toDateString() === new Date().toDateString());
    return hourlyList.map((item, index) => (
      <div key={index} className="hourly-item">
        <p className="hour">{new Date(item.dt * 1000).getHours()}:00</p>
        <img src={`icons/${item.weather[0].icon}.png`} className="icon-small" alt="weather" />
        <p className="temperature">{convertTemperature(item.main.temp)}°{unit}</p>
        <p className="description">{item.weather[0].description}</p>
      </div>
    ));
  };

  return (
    <div className="hourly-weather">
      <h2>Hourly Forecast</h2>
      <div className="hourly-list">
        {getHourlyData()}
      </div>
    </div>
  );
};

export default HourlyWeather;


.hourly-weather {
  margin: 20px;
}

.hourly-list {
  display: flex;
  flex-wrap: wrap;
}

.hourly-item {
  flex: 1 1 100px;
  text-align: center;
  margin: 10px;
}

.icon-small {
  width: 50px;
  height: 50px;
}

.temperature {
  font-size: 1.2rem;
}

.hour {
  font-size: 1rem;
  font-weight: bold;
}
